Example: Define PTI Code (PointOfIssueCode)

global function form_loaded
// make PTI field visible
dw_1.create=column=id=userfield1<tab>text=PTI<tab>name=pti<tab>x=1700<tab>y=640
end global
 
windowevent pretransfer
// set PTI field
setvar=e_pticode=<dw_1.item.pti.value>
end event
