Example:

setvar=ls_linetext=this is my new line
// Text line= 0 / Sum = 1
vdoc=speciallines=linetype=0 
// Afert first line
vdoc=speciallines=afterlinenumber=0 
vdoc=speciallines=linetext=<ls_linetext>
vdoc=speciallines=add
